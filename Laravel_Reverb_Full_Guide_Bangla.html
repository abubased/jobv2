১. সারসংক্ষেপ ও লক্ষ্য

লক্ষ্য ছিল application.submit-now.online-এর Laravel অ্যাপ অক্ষত রেখে wss://ws.submit-now.online দিয়ে Reverb চালু করা। ২০২৬-১১-২৩ তারিখে এটি অর্জিত হয়েছে: ব্রাউজার Console থেকে wss:// সংযোগ খুলেছে এবং Reverb pusher:connection_established বার্তা ফেরত দিয়েছে।

একটি Node.js লঞ্চার (app.js) ছাড়া কিছু যোগ করা হয়নি। এটি ws.submit-now.online-এর Passenger অ্যাপ হিসেবে চলে এবং তিনটি কাজ করে:

  • PHP CLI দিয়ে Laravel-এর artisan reverb:start চালু করে, একই প্রসেস দ্বিতীয়বার চালু হতে দেয় না।
  • বাইরে থেকে আসা HTTP ও WebSocket অনুরোধ ভেতরের 127.0.0.1:8080-এ পাঠায়।
  • /__diag ও /__health ঠিকানায় অবস্থা দেখায়, ফলে SSH বা Terminal ছাড়াই পরীক্ষা চলে।
Laravel অ্যাপ ও payment code-এ কোনো পরিবর্তন করা হয়নি। যা এখনো যাচাই হয়নি (ইভেন্ট broadcast, ফ্রন্টএন্ড কনফিগ ও চ্যানেল অথেনটিকেশন), তা শেষ অংশে তালিকাভুক্ত আছে।

২. সার্ভার ও কনফিগারেশন তথ্য

নিচের সব মান আপনার যাচাই-পত্র ও এই চ্যাটের পরীক্ষা থেকে নেওয়া। এই ডকুমেন্টে কোনো APP_KEY, REVERB_APP_SECRET বা পাসওয়ার্ড রাখা হয়নি।

৩. স্থাপত্য

ব্রাউজার ও Laravel দুজনেই ws.submit-now.online:443-এ অনুরোধ পাঠায়। Passenger অনুরোধটি app.js-কে দেয়, আর app.js সেটি 127.0.0.1:8080-এ চলা Reverb-এ পৌঁছে দেয়। Reverb বন্ধ থাকলে app.js প্রথমে সেটি চালু করে নেয়।

৫. app.js লঞ্চার

app.js একই সঙ্গে Reverb চালু করার লঞ্চার এবং wss:// অনুরোধ Reverb-এ পাঠানোর প্রক্সি। এটি /home/submitno/reverbapp/-এ থাকে এবং ws.submit-now.online-এর Passenger অ্যাপ হিসেবে চলে।

কীভাবে কাজ করে

  • PHP বাছাই: PHP_BIN, ea-php83, ea-php82, /usr/local/bin/php, /opt/alt/php83/usr/bin/php, /usr/bin/php ক্রমানুসারে পরীক্ষা করে। শুধু সেই PHP নেয় যার SAPI cli এবং ভার্সন 8.2 বা তার উপরে। তাই cgi-fcgi ভার্সন নিজে থেকেই বাদ পড়ে।
  • Laravel ফোল্ডার খোঁজা: LARAVEL_PATH, তারপর হোম ডিরেক্টরির application.submit-now.online, public_html/application.submit-now.online, public_html, application, laravel। যেখানে artisan ফাইল পায়, সেটিই নেয়।
  • Reverb চালু: Laravel ফোল্ডারে গিয়ে php artisan reverb:start --host=127.0.0.1 --port=8080 আলাদা (detached) প্রসেস হিসেবে চালায়।
  • Duplicate প্রতিরোধ: চালানোর আগে পোর্ট 8080 খোলা আছে কি না দেখে। reverb.pid লক ফাইলের পুরোনো PID জীবিত থাকলে অপেক্ষা করে, নতুন প্রসেস চালায় না। একসাথে একাধিক স্টার্টও আটকানো আছে।
  • প্রক্সি: HTTP অনুরোধ ও WebSocket upgrade দুটোই 127.0.0.1:8080-এ পাঠায়। অনুরোধ আসার সময় Reverb বন্ধ থাকলে আগে চালু করে নেয়।
  • লগ: reverb-launcher.log ফাইলে সব ঘটনা লেখে।

'use strict';
/**
 * Reverb Launcher + WebSocket Proxy for cPanel Passenger (Node.js)
 * Domain: ws.submit-now.online  ->  Reverb @ 127.0.0.1:8080
 *
 * শুধু নিচের CONFIG অংশ আপনার সার্ভার অনুযায়ী ঠিক করুন।
 */
const http = require('http');
const net = require('net');
const fs = require('fs');
const path = require('path');
const { spawn, execFileSync } = require('child_process');

// ===================== CONFIG =====================
const os = require('os');
const HOME = os.homedir();
const LARAVEL_CANDIDATES = [
  process.env.LARAVEL_PATH,
  path.join(HOME, 'application.submit-now.online'),
  path.join(HOME, 'public_html/application.submit-now.online'),
  path.join(HOME, 'public_html'),
  path.join(HOME, 'application'),
  path.join(HOME, 'laravel'),
].filter(Boolean);
function findLaravel() {
  for (const p of LARAVEL_CANDIDATES) {
    if (fs.existsSync(path.join(p, 'artisan'))) return p;
  }
  return null;
}
const REVERB_HOST = '127.0.0.1';
const REVERB_PORT = parseInt(process.env.REVERB_PORT || '8080', 10);
const PHP_CANDIDATES = [
  process.env.PHP_BIN,
  '/opt/cpanel/ea-php83/root/usr/bin/php', // artisan --version দিয়ে যাচাই করা (Laravel 12.39.0)
  '/opt/cpanel/ea-php82/root/usr/bin/php',
  '/usr/local/bin/php',
  '/opt/alt/php83/usr/bin/php',
  '/usr/bin/php', // cgi-fcgi হলে নিচে বাদ পড়বে
].filter(Boolean);
const LOG_FILE = process.env.REVERB_LOG || path.join(__dirname, 'reverb-launcher.log');
const LOCK_FILE = path.join(__dirname, 'reverb.pid');
// ==================================================

function log(msg) {
  const line = `[${new Date().toISOString()}] ${msg}\n`;
  try { fs.appendFileSync(LOG_FILE, line); } catch (_) {}
}

function findPhp() {
  for (const p of PHP_CANDIDATES) {
    try {
      if (!fs.existsSync(p)) continue;
      const out = execFileSync(p, ['-r', 'echo PHP_VERSION . "|" . php_sapi_name();'], { timeout: 5000 })
        .toString().trim();
      const [v, sapi] = out.split('|');
      const [major, minor] = v.split('.').map(Number);
      const okVersion = major > 8 || (major === 8 && minor >= 2); // Reverb-এর জন্য ৮.২+
      if (sapi === 'cli' && okVersion) return { bin: p, version: v, sapi };
      log(`PHP বাদ: ${p} (version ${v}, sapi ${sapi})`);
    } catch (e) {
      log(`PHP পরীক্ষা ব্যর্থ: ${p} - ${e.message}`);
    }
  }
  return null;
}

function isPortOpen(port, host = REVERB_HOST, timeout = 800) {
  return new Promise((resolve) => {
    const s = net.connect({ port, host });
    const done = (ok) => { s.destroy(); resolve(ok); };
    s.setTimeout(timeout);
    s.once('connect', () => done(true));
    s.once('timeout', () => done(false));
    s.once('error', () => done(false));
  });
}

function pidAlive(pid) {
  try { process.kill(pid, 0); return true; } catch (_) { return false; }
}

let starting = null; // একসাথে একাধিক স্টার্ট আটকানোর জন্য

async function ensureReverb() {
  if (await isPortOpen(REVERB_PORT)) return { running: true, started: false };
  if (starting) return starting;

  starting = (async () => {
    try {
      // লক ফাইলে থাকা পুরোনো PID এখনো জীবিত থাকলে নতুন প্রসেস চালাব না
      if (fs.existsSync(LOCK_FILE)) {
        const old = parseInt(fs.readFileSync(LOCK_FILE, 'utf8'), 10);
        if (old && pidAlive(old)) {
          log(`পুরোনো Reverb PID ${old} জীবিত, পোর্টের জন্য অপেক্ষা`);
          for (let i = 0; i < 10; i++) {
            if (await isPortOpen(REVERB_PORT)) return { running: true, started: false };
            await new Promise((r) => setTimeout(r, 500));
          }
        }
      }

      const php = findPhp();
      if (!php) {
        log('PHP 8.2 binary পাওয়া যায়নি');
        return { running: false, error: 'php_not_found' };
      }
      const LARAVEL_PATH = findLaravel();
      if (!LARAVEL_PATH) {
        log(`artisan পাওয়া যায়নি। যাচাই করা পথ: ${LARAVEL_CANDIDATES.join(', ')}`);
        return { running: false, error: 'artisan_not_found' };
      }

      const out = fs.openSync(LOG_FILE, 'a');
      const child = spawn(
        php.bin,
        ['artisan', 'reverb:start', `--host=${REVERB_HOST}`, `--port=${REVERB_PORT}`],
        { cwd: LARAVEL_PATH, detached: true, stdio: ['ignore', out, out] }
      );
      child.unref();
      fs.writeFileSync(LOCK_FILE, String(child.pid));
      log(`Reverb চালু হয়েছে: PID ${child.pid}, PHP ${php.version} (${php.bin})`);

      child.on('exit', (code) => log(`Reverb বন্ধ হয়েছে, exit code ${code}`));

      for (let i = 0; i < 20; i++) {
        if (await isPortOpen(REVERB_PORT)) return { running: true, started: true, pid: child.pid };
        await new Promise((r) => setTimeout(r, 500));
      }
      log('Reverb স্টার্ট হয়েছে কিন্তু পোর্ট খোলেনি (১০ সেকেন্ড)');
      return { running: false, error: 'port_timeout' };
    } catch (e) {
      log(`ensureReverb ত্রুটি: ${e.stack || e.message}`);
      return { running: false, error: e.message };
    } finally {
      starting = null;
    }
  })();
  return starting;
}

const server = http.createServer(async (req, res) => {
  if (req.url === '/__diag') {
    const php = findPhp();
    const report = {

      reverbPortOpen: await isPortOpen(REVERB_PORT),
      lastLog: (() => {
        try { return fs.readFileSync(LOG_FILE, 'utf8').split('\n').slice(-25); } catch (_) { return []; }
      })(),
    };
    res.writeHead(200, { 'Content-Type': 'application/json; charset=utf-8' });
    return res.end(JSON.stringify(report, null, 2));
  }

  if (req.url === '/__health') {
    const st = await ensureReverb();
    res.writeHead(st.running ? 200 : 503, { 'Content-Type': 'application/json' });
    return res.end(JSON.stringify({ app: 'reverb-launcher', node: process.version, reverb: st }));
  }

  const st = await ensureReverb();
  if (!st.running) {
    res.writeHead(503, { 'Content-Type': 'text/plain' });
    return res.end('Reverb is not running');
  }

  const proxyReq = http.request(
    { host: REVERB_HOST, port: REVERB_PORT, method: req.method, path: req.url, headers: req.headers },
    (proxyRes) => {
      res.writeHead(proxyRes.statusCode, proxyRes.headers);
      proxyRes.pipe(res);
    }
  );
  proxyReq.on('error', (e) => {
    log(`HTTP proxy ত্রুটি: ${e.message}`);
    if (!res.headersSent) res.writeHead(502);
    res.end('Bad gateway');
  });
  req.pipe(proxyReq);
});

// WebSocket (wss -> ws://127.0.0.1:8080)
server.on('upgrade', async (req, socket, head) => {
  const st = await ensureReverb();
  if (!st.running) {
    socket.end('HTTP/1.1 503 Service Unavailable\r\n\r\n');
    return;
  }
  const upstream = net.connect(REVERB_PORT, REVERB_HOST, () => {
    let raw = `${req.method} ${req.url} HTTP/1.1\r\n`;
    for (let i = 0; i < req.rawHeaders.length; i += 2) {
      raw += `${req.rawHeaders[i]}: ${req.rawHeaders[i + 1]}\r\n`;
    }
    raw += '\r\n';
    upstream.write(raw);
    if (head && head.length) upstream.write(head);
    socket.pipe(upstream);
    upstream.pipe(socket);
  });
  upstream.on('error', (e) => { log(`WS upstream ত্রুটি: ${e.message}`); socket.destroy(); });
  socket.on('error', () => upstream.destroy());
  socket.on('close', () => upstream.destroy());
});

৯. রক্ষণাবেক্ষণ, সতর্কতা ও বাকি যাচাই

WebSocket সংযোগ সফল হলেও কয়েকটি বিষয় এখনো যাচাই বাকি, আর কয়েকটি নিয়ম মেনে চলতে হবে।

সতর্কতা

  • একই Reverb প্রসেস একাধিকবার চালু করা যাবে না, এতে EADDRINUSE হতে পারে। app.js পোর্ট ও reverb.pid যাচাই করে এটি আটকায়।
  • Laravel .env, Reverb credentials, APP_KEY বা APay/payment code পরিবর্তন করা যাবে না, যতক্ষণ না প্রয়োজন যাচাই করা হয়।
  • Passenger-এর HTTP Test সফল হওয়া মানেই WebSocket সফল নয়। আসল wss:// সংযোগ পরীক্ষা করে নিশ্চিত হতে হবে।
  • যাচাই-পত্রে ও এই ডকুমেন্টে কোনো APP_KEY, REVERB_APP_SECRET, পাসওয়ার্ড বা ব্যক্তিগত credential রাখা হয়নি। ভবিষ্যতে সংযোজনেও রাখবেন না।

স্থায়িত্ব

  • Passenger idle হলে Node.js প্রসেস বন্ধ হতে পারে। Reverb আলাদা (detached) প্রসেস, তাই চলতে থাকে।
  • হোস্টিং দীর্ঘস্থায়ী প্রসেস বন্ধ করে দিলে, পরের WebSocket বা HTTP অনুরোধে app.js পোর্ট যাচাই করে Reverb আবার চালু করে। অনুরোধ না এলে এটি নিজে থেকে চালু হয় না।
  • কয়েক ঘণ্টা পর https://ws.submit-now.online/__health খুলে "running": true আছে কি না দেখে নিন।
  • বারবার বন্ধ হলে হোস্টিং প্রদানকারীকে জিজ্ঞেস করুন তারা background process চালাতে দেয় কি না।

এখনো যাচাই বাকি

  • সার্ভার থেকে ইভেন্ট broadcast। Laravel ইভেন্ট পাঠায় ws.submit-now.online:443-এ HTTP POST করে, যা এই প্রক্সির ভেতর দিয়ে যায়। অ্যাপে আসল ইভেন্ট ঘটিয়ে দেখুন। ব্যর্থ হলে storage/logs/laravel.log দেখুন।
  • ফ্রন্টএন্ড কনফিগ। .env-এ VITE_REVERB_HOST=ws.submit-now.online, VITE_REVERB_PORT=443, VITE_REVERB_SCHEME=https ও VITE_REVERB_APP_KEY আছে কি না দেখুন। এগুলো বদলালে npm run build আবার চালিয়ে নতুন asset আপলোড করতে হয়।
  • Private/Presence চ্যানেল। এগুলোতে অথেনটিকেশন লাগে, তাই /broadcasting/auth কাজ করছে কি না দেখুন।
  • Config cache। .env বদলালে php artisan config:clear লাগে। Terminal না থাকলে bootstrap/cache/config.php ফাইলটি (থাকলে) File Manager দিয়ে মুছলেও চলে।
  • /up পথের ফল। https://ws.submit-now.online/up খুলে Reverb-এর স্বাস্থ্য পরীক্ষার সাড়া দেখে এখানে নথিভুক্ত করুন।
